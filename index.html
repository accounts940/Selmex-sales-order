AUTO STATUS: when a dashboard user changes items, the order status follows (skipped for n8n) -- all items Dispatched -> Dispatched | all Done/Dispatched -> Completed | any started -> In Production | else Pending create or replace function public.recompute_order_status() returns trigger language plpgsql security definer set search_path = public as $$ declare v_order bigint; v_total int; v_disp int; v_fin int; v_started int; v_new text; v_cur text; v_comp timestamptz; begin if my_role() is null then return null; end if; -- n8n / service role: do nothing v_order := coalesce(new.order_id, old.order_id); select count(*), count(*) filter (where item_status = 'Dispatched'), count(*) filter (where item_status in ('Done','Dispatched')), count(*) filter (where coalesce(item_status,'Pending') <> 'Pending') into v_total, v_disp, v_fin, v_started from order_items where order_id = v_order; if v_total = 0 then return null; end if; v_new := case when v_disp = v_total then 'Dispatched' when v_fin = v_total then 'Completed' when v_started > 0 then 'In Production' else 'Pending' end; select status, completed_at into v_cur, v_comp from orders where id = v_order; if v_cur is distinct from v_new then update orders set status = v_new, completed_at = case when v_new in ('Completed','Dispatched') then coalesce(v_comp, now()) else null end, dispatched_at = case when v_new = 'Dispatched' then now() else null end where id = v_order; end if; return null; end $$; drop trigger if exists trg_recompute_status on order_items; create trigger trg_recompute_status after insert or update or delete on order_items for each row execute function recompute_order_status();
